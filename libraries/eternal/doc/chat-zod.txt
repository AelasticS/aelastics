  ZodObject,

  ZodUnion,
  ZodIntersection,
  ZodRecord,
  ZodMap,
  ZodSet,
  ZodLazy,
  ZodEffects,

export function cyclicSafe<T>(schema: ZodType<T>): ZodType<T> {
  class ZodCycleSafe extends ZodType<T> {
    _parse(input: ParseInput): ParseReturnType<T> {
      const seen = new WeakSet();

      const safeParse = (schema: ZodType<any>, data: any, path: (string | number)[]): ParseReturnType<any> => {
        if (typeof data === 'object' && data !== null) {
          if (seen.has(data)) {
            return { status: 'valid', value: data }; // Skip further recursion on already visited objects
          }
          seen.add(data);
        }

        const result = schema.safeParse(data);
        if (!result.success) return result;

        // recursively handle known nested schemas
        if (schema instanceof ZodObject) {
          for (const key in schema.shape) {
            const childResult = safeParse(schema.shape[key], data[key], [...path, key]);
            if (childResult.status === 'invalid') return childResult;
          }
        } else if (schema instanceof ZodArray) {
          for (let i = 0; i < data.length; i++) {
            const elementResult = safeParse(schema.element, data[i], [...path, i]);
            if (elementResult.status === 'invalid') return elementResult;
          }
        } else if (schema instanceof ZodUnion) {
          // Validate each possible option
          for (const option of schema.options) {
            const optionResult = safeParse(option, data, path);
            if (optionResult.status === 'valid') return elementResult;
          }
          return result; // if no valid union found
        } else if (schema instanceof ZodIntersection) {
          const leftResult = safeParse(schema._def.left, data, path);
          if (leftResult.status === 'invalid') return leftResult;
          const rightResult = safeParse(schema._def.right, data, path);
          if (rightResult.status === 'invalid') return rightResult;
        } else if (schema instanceof ZodLazy) {
          return safeParse(schema._def.getter(), data, path);
        } else if (schema instanceof ZodEffects) {
          return safeParse(schema.innerType(), data, path);
        }
        // Add similar logic for maps, sets, records if needed

        return result; // default for primitives
      };

      const seen = new WeakSet();
      return safeValidate(schema, input.data, []);
    };

    return schema.transform(data => data); // identity transform for compatibility
  }
}

import { ZodType } from 'zod';

export class Schema {
  public types: Map<string, ZodType>;
  public subschemas: Map<string, Schema>;

  constructor(public name: string, public parent?: Schema) {
    this.types = new Map();
    this.subschemas = new Map();
  }

  get fullPath(): string {
    return this.parent ? `${this.parent.fullPath}/${this.name}` : `/${this.name}`;
  }

  addType(typeName: string, schema: ZodType) {
    if (this.types.has(typeName)) {
      throw new Error(`Type "${typeName}" already exists in schema "${this.fullPath}"`);
    }
    this.types.set(typeName, schema);
  }

  addSubschema(subSchema: Schema): void {
    if (this.subschemas.has(subSchema.name)) {
      throw new Error(`Subschema "${subSchema.name}" already exists.`);
    }
    this.subschemas.set(subSchema.name, subSchema);
  }

  getType(typeName: string): ZodType | undefined {
    return this.types.get(typeName);
  }

  getSubschema(schemaName: string): Schema | undefined {
    return this.subschemas.get(schemaName);
  }
}

// SchemaRegistry.ts
import { Schema } from './Schema';

export class SchemaRegistry {
  private schemas = new Map<string, Schema>();

  registerSchema(schema: Schema): void {
    if (this.schemas.has(schema.fullPath)) {
      throw new Error(`Schema "${schema.fullPath}" already registered.`);
    }
    this.schemas.set(schema.fullPath, schema);
  }

  getSchema(fullPath: string): Schema | undefined {
    return this.schemas.get(fullPath);
  }

  unregisterSchema(schema: Schema): void {
    this.schemas.delete(schema.fullPath);
  }
}
